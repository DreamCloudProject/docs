export const iconFor = (name) => {
  if (iconMap[name]) return iconMap[name]

  const n = name.toLowerCase()

  if (n.includes("search")) return "magnifying-glass"
  if (n.includes("filter")) return "filter"
  if (n.includes("cart")) return "cart-shopping"
  if (n.includes("checkout")) return "credit-card"
  if (n.includes("order")) return "receipt"

  if (n.includes("login")) return "user"
  if (n.includes("logout")) return "right-from-bracket"
  if (n.includes("user") || n.includes("profile") || n.includes("account")) return "user"

  if (n.includes("language")) return "language"
  if (n.includes("switcher")) return "arrows-rotate"

  if (n.includes("modal") || n.includes("window")) return "window"

  if (n.includes("cookie") || n.includes("consent")) return "cookie-bite"
  if (n.includes("notification") || n.includes("alert")) return "bell"

  if (n.includes("radio")) return "circle-dot"
  if (n.includes("checkbox")) return "square-check"
  if (n.includes("select") || n.includes("dropdown") || n.includes("enumeration")) return "list"

  if (
    n.includes("text-field") ||
    n.includes("telephone") ||
    n.includes("password") ||
    n.includes("input") ||
    n.includes("form") ||
    n.includes("field")
  )
    return "i-cursor"

  if (n.includes("table") || n.includes("grid")) return "table"

  if (n.includes("image")) return "image"
  if (n.includes("video")) return "video"
  if (n.includes("upload")) return "upload"
  if (n.includes("download")) return "download"

  if (n.includes("link")) return "link"
  if (n.includes("menu")) return "bars"

  if (n.includes("quote")) return "quote-left"
  if (n.includes("rating") || n.includes("review")) return "star"

  if (n.includes("documents") || n.includes("article")) return "file"
  if (n.includes("tabs")) return "folder"
  if (n.includes("process")) return "diagram-project"

  if (n.includes("web3") || n.includes("wallet") || n.includes("nft")) return "wallet"

  return "cube"
}

export const descriptionFor = (name) => {
  const n = name.toLowerCase()

  if (n === "msm-yookassa-widget") return "Виджет оплаты YooKassa."
  if (n === "msm-cdek-widget") return "Виджет доставки CDEK."
  if (n.includes("web3") || n.includes("wallet") || n.includes("nft")) return "Web3 UI (кошелёк / NFT)."

  if (n.includes("search")) return "UI поиска (ввод, результаты, фильтры, пагинация)."
  if (n.includes("filter")) return "UI фильтрации для списков/поиска."

  if (n.includes("checkout")) return "UI и флоу checkout."
  if (n.includes("cart")) return "UI корзины."
  if (n.includes("order")) return "UI заказов (детали, история)."

  if (n.includes("login") || n.includes("logout") || n.includes("reset-password")) return "UI авторизации."
  if (n.includes("user") || n.includes("profile") || n.includes("account")) return "UI личного кабинета."

  if (n.includes("cookie") || n.includes("consent")) return "UI cookie-согласия / настроек."
  if (n.includes("notification") || n.includes("alert")) return "UI уведомлений / алертов."

  if (n.includes("upload") || n.includes("documents")) return "UI загрузки (upload)."

  if (n.includes("image") || n.includes("video")) return "UI медиа (изображения/видео)."
  if (n.includes("table") || n.includes("grid")) return "UI таблиц / data grid."

  if (n.includes("menu") || n.includes("breadcrumbs") || n.includes("link")) return "UI навигации."
  if (n.includes("tabs")) return "UI вкладок (tabs)."

  if (n.includes("row") || n.includes("column") || n.includes("container")) return "UI раскладки / контейнеров."

  if (
    n.includes("form") ||
    n.includes("field") ||
    n.includes("checkbox") ||
    n.includes("radio") ||
    n.includes("select") ||
    n.includes("dropdown") ||
    n.includes("enumeration") ||
    n.includes("text-field") ||
    n.includes("telephone") ||
    n.includes("password") ||
    n.includes("input")
  )
    return "UI форм и полей ввода."

  if (n.includes("rating") || n.includes("review")) return "UI рейтинга / отзывов."
  if (n.includes("process")) return "UI шагов / процесса."

  if (n.includes("blog") || n.includes("article") || n.includes("story")) return "UI контента (статьи/истории)."
  if (n.includes("banner") || n.includes("logo")) return "UI баннеров / промо."

  if (n.includes("ssr")) return "UI помощник для SSR."

  return "UI-компонент."
}
